cmake_minimum_required(VERSION 3.16)
project(fsi2_mesquite LANGUAGES CXX)
set(MESQUITE_ROOT "$ENV{HOME}/.local/mesquite/2.3.0" CACHE PATH "Mesquite installation prefix")
find_path(MESQUITE_INCLUDE_DIR Mesquite_all_headers.hpp HINTS "${MESQUITE_ROOT}/include" REQUIRED NO_DEFAULT_PATH)
find_library(MESQUITE_LIBRARY NAMES mesquite HINTS "${MESQUITE_ROOT}/lib" "${MESQUITE_ROOT}/lib64" REQUIRED NO_DEFAULT_PATH)
add_library(mesquite_smoothing SHARED mesquite_smoothing.cpp)
target_compile_features(mesquite_smoothing PRIVATE cxx_std_11)
target_include_directories(mesquite_smoothing PRIVATE "${MESQUITE_INCLUDE_DIR}")
target_link_libraries(mesquite_smoothing PRIVATE "${MESQUITE_LIBRARY}")
if(UNIX)
  target_link_libraries(mesquite_smoothing PRIVATE m)
endif()
